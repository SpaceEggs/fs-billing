table inet filter {
	set ssh_rate_v4 {
		type ipv4_addr
		size 65535
		flags dynamic
		elements = { <工作站_IP> limit rate 6/minute burst 4 packets }
	}

	set ssh_rate_v6 {
		type ipv6_addr
		size 65535
		flags dynamic
	}

	chain input {
		type filter hook input priority filter; policy drop;
		iif "lo" accept
		ct state established,related accept
		ct state invalid drop
		ip protocol icmp accept
		ip6 nexthdr ipv6-icmp accept
		tcp dport 22 ct state new add @ssh_rate_v4 { ip saddr limit rate 6/minute burst 4 packets } accept
		tcp dport 22 ct state new add @ssh_rate_v6 { ip6 saddr limit rate 6/minute burst 4 packets } accept
		tcp dport 22 drop
		tcp dport { 80, 443 } accept
		ip saddr <A_占位> tcp dport 3001 accept
	}

	chain output {
		type filter hook output priority filter; policy accept;
	}

	chain forward {
		type filter hook forward priority filter; policy drop;
	}
}
